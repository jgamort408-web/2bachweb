Sean las matrices
\[
A=\begin{pmatrix}1&-1&0\\0&1&-1\end{pmatrix},\quad
B=\begin{pmatrix}1&0\\0&1\\2&-2\end{pmatrix}\quad\text{y}\quad
C=\begin{pmatrix}1&1\\3&-2\end{pmatrix}.
\]
\begin{enumerate}[label=\alph*)]
\item \textbf{(1.2 puntos)} Razone cuáles de las siguientes operaciones son posibles:
\[
A\cdot B^t\qquad B+3C\qquad C\cdot B^t\qquad A\cdot B+C
\]
\item \textbf{(1.3 puntos)} Resuelva la ecuación matricial $A\cdot B\cdot X=C$
\end{enumerate}
