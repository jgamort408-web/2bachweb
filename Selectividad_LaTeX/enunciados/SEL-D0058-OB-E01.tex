\item \textbf{(1.25 puntos)} Efectúe, si es posible, los siguientes productos:
\[
A\cdot A^t;\quad A^t\cdot A;\quad A\cdot B
\]
\item \textbf{(1.25 puntos)} Resuelva la siguiente ecuación matricial $A\cdot A^t\cdot X=B$
\end{enumerate}
