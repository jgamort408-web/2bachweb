Sean las matrices
\[
A=\begin{pmatrix}\dfrac15&0\\[0.7em]-\dfrac25&\dfrac35\end{pmatrix},\quad
B=\begin{pmatrix}\dfrac35&-1\\[0.7em]\dfrac45&\dfrac45\end{pmatrix},\quad
C=\begin{pmatrix}1&0&-1\\2&1&3\end{pmatrix}.
\]
\begin{enumerate}[label=\alph*)]
\item \textbf{(1.5 puntos)} Resuelva la ecuación matricial $(2A+B)\cdot X=3A-B$.
\item \textbf{(1 punto)} Determine en cada caso la dimensión de la matriz $D$ para que se puedan realizar las siguientes operaciones:
\[
C\cdot D+A,\quad C^t\cdot D\cdot C,\quad D\cdot C^t,\quad C\cdot D\cdot C^t.
\]
\end{enumerate}
