Sean las matrices $A=\begin{pmatrix}2&1\\3&-2\end{pmatrix}$ y $B=\begin{pmatrix}3&-2\\1&4\end{pmatrix}$.
\begin{enumerate}[label=\alph*)]
\item \textbf{(0.75 puntos)} Efectúe la operación $A\cdot B^t$.
\item \textbf{(0.75 puntos)} Determine la matriz $X$ tal que $A+2\cdot X=B$.
\item \textbf{(1 punto)} Halle la matriz $Y$ tal que $B\cdot Y=\begin{pmatrix}6\\9\end{pmatrix}$.
\end{enumerate}
