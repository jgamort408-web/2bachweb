Sean las matrices $A=\begin{pmatrix}1&-1\\0&2\end{pmatrix}$, $B=\begin{pmatrix}3&1\\-1&1\end{pmatrix}$.
\begin{enumerate}[label=\alph*)]
\item \textbf{(1 punto)} Calcule $A^2$ y $2B+I_2$.
\item \textbf{(2 puntos)} Resuelva la ecuación matricial $A\cdot X-I_2=2B^2$.
\end{enumerate}
