\begin{enumerate}[label=\alph*)]
\item \textbf{(0.5 puntos)} Si $A$ es una matriz de dimensión $m\times n$, indique la dimensión de una matriz $X$ si se verifica que $(A^t\cdot A)\cdot X=I_n$.
\item \textbf{(1.25 puntos)} Calcule dicha matriz $X$ en el caso en que
\[
A=\begin{pmatrix}1&1\\1&-1\\1&1\end{pmatrix}.
\]
\item \textbf{(0.75 puntos)} Calcule, si es posible, el producto $A\cdot(A^t\cdot A)$.
\end{enumerate}
