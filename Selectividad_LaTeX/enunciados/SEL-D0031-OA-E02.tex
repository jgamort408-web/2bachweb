\begin{enumerate}[label=\alph*)]
\item \textbf{(1.2 puntos)} Calcule los valores de $a$ y $b$ para que la función
\[
f(x)=\begin{cases}
\dfrac{b}{2-x} & \text{si }x\leq1\\[4pt]
ax^2-3x+1 & \text{si }x>1
\end{cases}
\]
sea derivable en el punto de abscisa $x=1$.
\item \textbf{(1.3 puntos)} Para $a=1$ y $b=2$, estudie su monotonía y determine las ecuaciones de sus asíntotas, si existen.
\end{enumerate}
