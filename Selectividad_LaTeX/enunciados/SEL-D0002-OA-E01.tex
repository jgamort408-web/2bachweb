Se consideran las matrices
\[
  A=\begin{pmatrix}1&0&-2\\-1&1&0\end{pmatrix}
  \qquad
  B=\begin{pmatrix}1&-3\\-2&0\end{pmatrix}
  \qquad
  C=\begin{pmatrix}7&-12&16\\-1&7&12\end{pmatrix}
\]

\begin{enumerate}[label=\alph*)]
  \item \textbf{(1 punto)} Justifique cuáles de las siguientes afirmaciones
  son ciertas:
  \begin{enumerate}[label=\arabic*)]
    \item $A\cdot A^{t}$ es una matriz simétrica.
    \item $A\cdot A^{t}+B$ posee inversa.
  \end{enumerate}
  \item \textbf{(1.5 puntos)} Resuelva la ecuación matricial $B\cdot X+A=C$.
\end{enumerate}
