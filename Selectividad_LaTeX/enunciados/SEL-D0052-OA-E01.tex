Sean las matrices
\[
A=\begin{pmatrix}2&3\\3&5\end{pmatrix},\quad
B=\begin{pmatrix}3&-5&3\\0&2&1\end{pmatrix},\quad
C=\begin{pmatrix}8\\3\\0\end{pmatrix},\quad
D=\begin{pmatrix}5\\3\end{pmatrix}.
\]
\begin{enumerate}[label=\alph*)]
\item \textbf{(1 punto)} Calcule $A^3$.
\item \textbf{(1.5 puntos)} Determine la matriz $X$ para que $A\cdot X+B\cdot C=D$.
\end{enumerate}
