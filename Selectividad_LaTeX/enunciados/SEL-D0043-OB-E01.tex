\begin{enumerate}[label=\alph*)]
\item \textbf{(1 punto)} Determine los valores de $x$ e $y$ que hacen cierta la igualdad
\[
\begin{pmatrix}2&-1\\3&-1\end{pmatrix}\cdot\begin{pmatrix}x\\-y\end{pmatrix}=\begin{pmatrix}1&x\\y&-1\end{pmatrix}\cdot\begin{pmatrix}3\\0\end{pmatrix}.
\]
\item \textbf{(1.5 puntos)} Resuelva la ecuación matricial:
\[
X\cdot\begin{pmatrix}1&3\\2&5\end{pmatrix}-2\cdot\begin{pmatrix}0&-1\\-1&0\end{pmatrix}=\begin{pmatrix}1&2\\3&-1\end{pmatrix}.
\]
\end{enumerate}
