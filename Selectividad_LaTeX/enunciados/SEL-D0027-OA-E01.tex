Sean las matrices
\[
A=\begin{pmatrix}2&1\\0&-1\end{pmatrix},\quad B=\begin{pmatrix}1&-1\\2&0\end{pmatrix},\quad C=\begin{pmatrix}-2&4\\1&-1\end{pmatrix}\quad\text{y}\quad D=\begin{pmatrix}1&0&1\\0&1&0\end{pmatrix}.
\]
\begin{enumerate}[label=\alph*)]
\item \textbf{(1 punto)} Razone si se pueden efectuar las siguientes operaciones
\[
A\cdot D+B\cdot C\qquad D^t\cdot B-A^2
\]
\item \textbf{(1.5 puntos)} Halle la matriz $X$ que verifica la ecuación matricial $A\cdot X=B-C$.
\end{enumerate}
