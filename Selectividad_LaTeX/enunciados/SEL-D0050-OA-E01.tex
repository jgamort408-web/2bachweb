Sean las matrices
\[
A=\begin{pmatrix}2&-1\\a&b\end{pmatrix}\quad\text{y}\quad
B=\begin{pmatrix}-1&1\\3&0\end{pmatrix}.
\]
\begin{enumerate}[label=\alph*)]
\item \textbf{(1.25 puntos)} Obtenga $a$ y $b$ sabiendo que $A^2=\begin{pmatrix}5&-2\\-2&1\end{pmatrix}$. \textquestiondown{}Es $A$ simétrica?
\item \textbf{(1.25 puntos)} Para los valores $a=3$ y $b=1$ calcule la matriz $X$ tal que $A\cdot B=2(X-3I_2)$.
\end{enumerate}
