Sea la función
\[
f(x)=\begin{cases}
\dfrac{1}{2}(ax-12) & \text{si }x<-1\\[4pt]
-x^2+b(x-1) & \text{si }x\geq-1
\end{cases}
\]
\begin{enumerate}[label=\alph*)]
\item \textbf{(1.5 puntos)} Halle los valores de $a$ y $b$ sabiendo que la función es derivable en $x=-1$.
\item \textbf{(1 punto)} Para $a=1$ y $b=-1$ obtenga la ecuación de la recta tangente a la gráfica de la función $f(x)$ en el punto de abscisa $x=-2$.
\end{enumerate}
