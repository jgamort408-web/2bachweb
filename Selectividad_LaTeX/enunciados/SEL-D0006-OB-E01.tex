Se consideran las matrices
\[
  A=\begin{pmatrix}1&1&3\\1&2&-1\\1&-1&-1\end{pmatrix}
  \qquad B=\begin{pmatrix}2\\3\\0\end{pmatrix}
\]
\begin{enumerate}[label=\alph*)]
  \item \textbf{(1.5 puntos)} Justifique que la matriz $A$ tiene inversa y
  calcule $A^{-1}$.
  \item \textbf{(1 punto)} Calcule, si existe, la matrix $X$ que satisface
  la ecuación matricial $A\cdot X=B$.
\end{enumerate}
