Sean las matrices $A=\begin{pmatrix}1&-7\\2&-1\end{pmatrix}$ y $B=\begin{pmatrix}1&0\\-5&2\end{pmatrix}$.
\begin{enumerate}[label=\alph*)]
\item \textbf{(1.25 puntos)} Calcule las matrices $X$ e $Y$ para las que se verifica
\[
X+Y=A\qquad\text{y}\qquad3X+Y=B.
\]
\item \textbf{(1.25 puntos)} Halle la matriz $Z$ que verifica $B\cdot Z+B^t=2I_2$.
\end{enumerate}
