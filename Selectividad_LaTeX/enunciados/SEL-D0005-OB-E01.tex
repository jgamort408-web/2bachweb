Se consideran las matrices
\[
  A=\begin{pmatrix}0&1&-1\\2&-1&0\end{pmatrix}
  \qquad B=\begin{pmatrix}2&0\\-2&1\\0&1\end{pmatrix}
  \qquad C=\begin{pmatrix}1&2\\2&-3\end{pmatrix}
\]
\begin{enumerate}[label=\alph*)]
  \item \textbf{(1.5 puntos)} \textquestiondown{}Tiene inversa la matriz
  $A\cdot B-C$? Justifique la respuesta y, en caso afirmativo, calcule
  $(A\cdot B-C)^{-1}$.
  \item \textbf{(1 punto)} Resuelva la ecuación matricial
  $A\cdot B\cdot X-C\cdot X=C^t$.
\end{enumerate}
