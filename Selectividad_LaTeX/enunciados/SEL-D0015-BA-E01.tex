Sean $A$, $B$, $X$, $Y$ matrices invertibles que verifican
$A\cdot X=B$ y $B\cdot Y=A$.
\begin{enumerate}[label=\alph*)]
  \item \textbf{(1 punto)} Compruebe que $Y^{-1}=X$.
  \item \textbf{(1.5 puntos)} Para
  $A=\begin{pmatrix}1&2\\1&3\end{pmatrix}$ y
  $B=\begin{pmatrix}2&1\\0&-1\end{pmatrix}$, halle $X$ e $Y$.
\end{enumerate}
