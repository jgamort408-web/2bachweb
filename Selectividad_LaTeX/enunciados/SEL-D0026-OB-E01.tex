Sean las matrices
\[
A=\begin{pmatrix}3&0\\1&2\end{pmatrix},\quad B=\begin{pmatrix}-2&3\end{pmatrix}\quad\text{y}\quad C=\begin{pmatrix}-1\\-1\end{pmatrix}.
\]
\begin{enumerate}[label=\alph*)]
\item \textbf{(1 punto)} Justifique cuáles de las siguientes operaciones se pueden realizar y en dichos casos calcule el resultado: $A\cdot B$, $B\cdot A$, $B\cdot C$ y $C^t\cdot B^t$.
\item \textbf{(1.5 puntos)} Calcule la matriz $X$ en la ecuación $A\cdot X+B^t=4C$.
\end{enumerate}
