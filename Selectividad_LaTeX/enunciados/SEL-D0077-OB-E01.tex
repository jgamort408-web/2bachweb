\begin{enumerate}[label=\alph*)]
\item \textbf{(2 puntos)} Halle la matriz $X$ que verifica la ecuación
\[
X\cdot\begin{pmatrix}2&5\\1&3\end{pmatrix}=\begin{pmatrix}1\\2\end{pmatrix}\cdot\begin{pmatrix}2&4\end{pmatrix}.
\]
\item \textbf{(1 punto)} Determine los valores de $x$ e $y$ que cumplen la igualdad
\[
\begin{pmatrix}1&0\\3&-1\end{pmatrix}\cdot\begin{pmatrix}x\\y\end{pmatrix}=\begin{pmatrix}2&1\\-x&y\end{pmatrix}\cdot\begin{pmatrix}1\\1\end{pmatrix}.
\]
\end{enumerate}
