Sean las matrices
\[
A=\begin{pmatrix}2&-5\\1&-3\end{pmatrix},\quad
B=\begin{pmatrix}3&-1&2\\0&1&1\end{pmatrix},\quad
C=\begin{pmatrix}1&2&3\\-1&5&3\end{pmatrix}.
\]
\begin{enumerate}[label=\alph*)]
\item \textbf{(1 punto)} Calcule $A^2-B\cdot C^t$.
\item \textbf{(1.5 puntos)} Resuelva la ecuación matricial $A\cdot X+B=2\cdot C$.
\end{enumerate}
