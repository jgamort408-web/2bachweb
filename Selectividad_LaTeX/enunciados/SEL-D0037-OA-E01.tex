Sean las matrices
\[
A=\begin{pmatrix}1&-1&2\\0&1&-1\\1&0&2\end{pmatrix},\quad
B=\begin{pmatrix}1&2&1\\1&-2&0\end{pmatrix},\quad
C=\begin{pmatrix}2&1\end{pmatrix},\quad
D=\begin{pmatrix}1&-1&2\end{pmatrix}.
\]
\begin{enumerate}[label=\alph*)]
\item \textbf{(0.8 puntos)} Estudie cuáles de los siguientes productos de matrices se pueden realizar, indicando las dimensiones de la matriz resultante:
\[
A\cdot B^t\qquad C^t\cdot D\qquad B^t\cdot D\qquad D\cdot B^t.
\]
\item \textbf{(0.5 puntos)} Despeje la matriz $X$ en la ecuación $X\cdot A^{-1}+2B=3C^t\cdot D$, sin calcular sus elementos.
\item \textbf{(1.2 puntos)} Calcule la matriz $A\cdot\left(B^t-2D^t\cdot C\right)$.
\end{enumerate}
