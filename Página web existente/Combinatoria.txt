<section id="comb">
  <div class="sec-header">
    <div class="sec-icon" style="background:var(--comb-bg)">🎯</div>
    <div>
      <div class="sec-title" style="color:var(--comb)">Combinaciones</div>
      <div class="sec-sub">Elegimos \(r\) elementos de \(n\) — el orden NO importa — sin repetición</div>
    </div>
  </div>

  <div class="card-grid">
    <div>
      <div class="formula-box fb-comb">
        <h3>Combinaciones \(C(n,r)=\dbinom{n}{r}\)</h3>
        \[ C(n,r) = \binom{n}{r} = \frac{n!}{r!\,(n-r)!} \]
        <div class="formula-note">Propiedad de simetría: \(C(n,r)=C(n,n-r)\)</div>
      </div>
      <div class="tip-box" style="background:#fdf4ff;border-color:var(--comb-bg)">
        <span class="tip-icon">💡</span>
        <p><strong>Relación con variaciones:</strong><br>
        \(\displaystyle C(n,r)=\frac{V(n,r)}{r!}\)<br>
        Dividimos entre \(r!\) porque el orden dentro del grupo no cuenta.</p>
      </div>
    </div>
    <div class="card">
      <h4 style="font-weight:700;margin-bottom:.8rem;color:var(--comb)">🎬 ¿Por qué dividimos entre \(r!\)?</h4>
      <p style="font-size:.85rem;color:var(--text-muted);margin-bottom:.7rem">Al elegir 2 de \(\{A,B,C\}\) sin que importe el orden:</p>
      <div style="display:grid;grid-template-columns:1fr 1fr;gap:.5rem;font-size:.88rem">
        <div style="background:#e9d5ff;border-radius:8px;padding:.5rem;text-align:center;color:var(--comb);font-weight:700">\(\{A,B\}=\{B,A\}\)<br><small style="opacity:.7">¡Mismo grupo!</small></div>
        <div style="background:#e9d5ff;border-radius:8px;padding:.5rem;text-align:center;color:var(--comb);font-weight:700">\(\{A,C\}=\{C,A\}\)<br><small style="opacity:.7">¡Mismo grupo!</small></div>
        <div style="background:#e9d5ff;border-radius:8px;padding:.5rem;text-align:center;color:var(--comb);font-weight:700">\(\{B,C\}=\{C,B\}\)<br><small style="opacity:.7">¡Mismo grupo!</small></div>
        <div style="background:#f5f0ff;border-radius:8px;padding:.5rem;text-align:center;color:var(--comb);font-weight:700;border:1.5px solid var(--comb-bg)">\(C(3,2)=3\)<br><small style="opacity:.7">Solo 3 grupos distintos</small></div>
      </div>
    </div>
  </div>

  <div class="card">
    <h4 style="font-weight:700;margin-bottom:.8rem;color:var(--comb);font-family:'Playfair Display',serif;font-size:1.1rem">🔺 Triángulo de Pascal — \(\binom{n}{r}\)</h4>
    <p style="font-size:.88rem;color:var(--text-muted);margin-bottom:1rem">Cada celda vale \(C(n,r)\). Pasa el ratón para ver la notación. Cada número es la suma de los dos superiores.</p>
    <div id="pascal-triangle" style="text-align:center;overflow-x:auto;padding:.5rem"></div>
  </div>

  <div class="example-box">
    <h4>📌 Ejemplo resuelto</h4>
    <p>De una clase de <strong>10 alumnos</strong> se elige un <strong>grupo de 3</strong> para representar al colegio. ¿Cuántos grupos posibles hay?</p>
    <p style="margin-top:.5rem">No importa quién es 1.º, 2.º o 3.º → \(C(10,3)\)</p>
    <button class="toggle-steps-btn" onclick="toggleSteps('steps-comb1')">▼ Ver paso a paso</button>
    <div class="steps-container" id="steps-comb1">
      <div class="step"><div class="step-num">1</div><div class="step-content">
        <div class="step-label">Datos</div>
        <div class="step-val">\(n=10,\quad r=3\)</div>
      </div></div>
      <div class="step"><div class="step-num">2</div><div class="step-content">
        <div class="step-label">Fórmula</div>
        <div class="step-val">\(\displaystyle C(10,3)=\frac{10!}{3!\cdot7!}\)</div>
      </div></div>
      <div class="step"><div class="step-num">3</div><div class="step-content">
        <div class="step-label">Simplificamos</div>
        <div class="step-val">\(\displaystyle=\frac{10\cdot9\cdot8\cdot\cancel{7!}}{3!\cdot\cancel{7!}}=\frac{10\cdot9\cdot8}{3\cdot2\cdot1}=\frac{720}{6}\)</div>
      </div></div>
      <div class="step"><div class="step-num">4</div><div class="step-content">
        <div class="step-label">Resultado</div>
        <div class="step-val">\(C(10,3)=\mathbf{120}\) grupos distintos</div>
      </div></div>
    </div>
    <div class="answer-tag">✅ \(C(10,3)=120\)</div>
  </div>
</section>

function calculate(){
  const t=document.getElementById('calc-type').value;
  const n=parseInt(document.getElementById('calc-n').value);
  const r=parseInt(document.getElementById('calc-r').value)||0;
  const repsRaw=document.getElementById('calc-reps').value;
  const resDiv=document.getElementById('calc-result');
  const resMain=document.getElementById('res-main');
  const resFormula=document.getElementById('res-formula');
  const resSteps=document.getElementById('res-steps');
  if(isNaN(n)||n<1||n>20){alert('n debe estar entre 1 y 20');return;}

  let result,steps=[],formula='';

  if(t==='vs'){
    if(r>n){alert('r no puede ser mayor que n');return;}
    result=factorial(n)/factorial(n-r);
    const fac=Array.from({length:r},(_,i)=>n-i).join('\\cdot');
    formula=`\\(V(${n},${r})=\\dfrac{${n}!}{${n-r}!}=${fac}=${fmt(result)}\\)`;
    steps=[
      `Fórmula: \\(V(n,r)=\\dfrac{n!}{(n-r)!}\\)`,
      `Sustituimos: \\(V(${n},${r})=\\dfrac{${n}!}{${n-r}!}\\)`,
      `Desarrollo: \\(${fac}=\\mathbf{${fmt(result)}}\\)`,
    ];
  } else if(t==='vr'){
    result=Math.pow(n,r);
    formula=`\\(VR(${n},${r})=${n}^{${r}}=${fmt(result)}\\)`;
    steps=[
      `Fórmula: \\(VR(n,r)=n^r\\)`,
      `Sustituimos: \\(VR(${n},${r})=${n}^{${r}}\\)`,
      `Resultado: \\(\\mathbf{${fmt(result)}}\\)`,
    ];
  } else if(t==='perm'){
    result=factorial(n);
    const fac=Array.from({length:n},(_,i)=>n-i).join('\\cdot');
    formula=`\\(P_{${n}}=${n}!=${fac}=${fmt(result)}\\)`;
    steps=[
      `Fórmula: \\(P_n=n!\\)`,
      `\\(P_{${n}}=${n}!=${fac}\\)`,
      `Resultado: \\(\\mathbf{${fmt(result)}}\\)`,
    ];
  } else if(t==='permrep'){
    const reps=repsRaw.split(',').map(s=>parseInt(s.trim())).filter(x=>!isNaN(x)&&x>0);
    if(!reps.length){alert('Introduce las repeticiones');return;}
    const sumReps=reps.reduce((a,b)=>a+b,0);
    if(sumReps>n){alert('La suma de repeticiones no puede superar n');return;}
    const denom=reps.reduce((a,b)=>a*factorial(b),1);
    result=factorial(n)/denom;
    const repStr=reps.map(x=>`${x}!`).join('\\cdot');
    formula=`\\(P_{${n}}^{(${reps.join(',')})}=\\dfrac{${n}!}{${repStr}}=\\dfrac{${fmt(factorial(n))}}{${fmt(denom)}}=${fmt(result)}\\)`;
    steps=[
      `Fórmula: \\(P_n^{(a,b,\\ldots)}=\\dfrac{n!}{a!\\cdot b!\\cdots}\\)`,
      `Numerador: \\(${n}!=${fmt(factorial(n))}\\)`,
      `Denominador: \\(${repStr}=${fmt(denom)}\\)`,
      `Resultado: \\(\\dfrac{${fmt(factorial(n))}}{${fmt(denom)}}=\\mathbf{${fmt(result)}}\\)`,
    ];
  } else if(t==='comb'){
    if(r>n){alert('r no puede ser mayor que n');return;}
    result=factorial(n)/(factorial(r)*factorial(n-r));
    const numFac=Array.from({length:r},(_,i)=>n-i).join('\\cdot');
    const numVal=Array.from({length:r},(_,i)=>n-i).reduce((a,b)=>a*b,1);
    formula=`\\(C(${n},${r})=\\dbinom{${n}}{${r}}=\\dfrac{${n}!}{${r}!\\cdot${n-r}!}=\\dfrac{${numFac}}{${r}!}=\\dfrac{${fmt(numVal)}}{${fmt(factorial(r))}}=${fmt(result)}\\)`;
    steps=[
      `Fórmula: \\(C(n,r)=\\dfrac{n!}{r!\\,(n-r)!}\\)`,
      `\\(C(${n},${r})=\\dfrac{${n}!}{${r}!\\cdot${n-r}!}\\)`,
      `Numerador (\\(${r}\\) factores): \\(${numFac}=${fmt(numVal)}\\)`,
      `Denominador: \\(${r}!=${fmt(factorial(r))}\\)`,
      `Resultado: \\(\\mathbf{${fmt(result)}}\\)`,
    ];
  }

  resFormula.innerHTML=formula;
  resMain.textContent=fmt(result);
  resSteps.innerHTML=steps.map(s=>`<div class="calc-step">${s}</div>`).join('');
  resDiv.classList.add('show');
  resDiv.animate([{opacity:0,transform:'translateY(10px)'},{opacity:1,transform:'translateY(0)'}],{duration:300,fill:'forwards'});
  typeset(resDiv);
}

function shuffle(a){return[...a].sort(()=>Math.random()-.5)}
function showPanel(id){['game-start','game-play','game-over'].forEach(p=>{document.getElementById(p).classList.toggle('active',p===id)})}
function showGameStart(){showPanel('game-start')}

function startGame(){
  GS={q:0,pts:0,correct:0,streak:0,questions:shuffle(QS)};
  showPanel('game-play');renderQ();
}

function renderQ(){
  const q=GS.questions[GS.q];
  document.getElementById('g-qnum').textContent=`${GS.q+1}/10`;
  document.getElementById('g-pts').textContent=GS.pts;
  document.getElementById('g-correct').textContent=GS.correct;
  document.getElementById('g-streak').textContent=`${GS.streak}🔥`;
  document.getElementById('g-progress').style.width=`${(GS.q/10)*100}%`;
  document.getElementById('g-label').textContent=q.cat;
  document.getElementById('g-text').textContent=q.q;
  document.getElementById('g-feedback').classList.remove('show','ok','bad');
  document.getElementById('g-next').style.display='none';

  const od=document.getElementById('g-opts');od.innerHTML='';
  shuffle(q.opts.map((o,i)=>({o,i}))).forEach(({o,i})=>{
    const b=document.createElement('button');b.className='game-opt-btn';
    // Render fractions with MathJax
    if(/^\d+\/\d+$/.test(o)){
      const[num,den]=o.split('/');
      b.innerHTML=`\\(\\dfrac{${num}}{${den}}\\)`;
    } else {
      b.textContent=o;
    }
    b.dataset.idx=i;
    b.onclick=()=>selectAns(b,i===q.ans,q,od);
    od.appendChild(b);
  });
  typeset(od);
}

function selectAns(btn,isOk,q,od){
  od.querySelectorAll('.game-opt-btn').forEach(b=>b.disabled=true);
  btn.classList.add(isOk?'correct':'wrong');
  if(!isOk){
    od.querySelectorAll('.game-opt-btn').forEach(b=>{if(parseInt(b.dataset.idx)===q.ans)b.classList.add('correct')});
  }
  if(isOk){GS.streak++;GS.pts+=GS.streak>=3?15:10;GS.correct++;}
  else GS.streak=0;

  const fb=document.getElementById('g-feedback');
  fb.innerHTML=`<strong>${isOk?'✅ ¡Correcto!':'❌ Incorrecto'}</strong> &mdash; ${q.exp}`;
  fb.classList.add('show',isOk?'ok':'bad');
  typeset(fb);
  document.getElementById('g-pts').textContent=GS.pts;
  document.getElementById('g-correct').textContent=GS.correct;
  document.getElementById('g-streak').textContent=`${GS.streak}🔥`;
  const nb=document.getElementById('g-next');
  nb.textContent=GS.q<9?'Siguiente pregunta →':'Ver resultados 🏁';
  nb.style.display='';
}

}
