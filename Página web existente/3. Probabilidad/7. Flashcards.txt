            { category: "Conceptos Básicos", icon: "🔗", question: "¿Qué son Sucesos Independientes?", answer: "Cuando lo que pase en uno no afecta al otro. Se cumple que: $$P(A \\cap B) = P(A) \\cdot P(B)$$" },
            { category: "Conceptos Básicos", icon: "🌓", question: "Suceso Contrario ($A^c$ o $\\bar{A}$)", answer: "Lo que falta para llegar al total. $$P(\\bar{A}) = 1 - P(A)$$" },

            // --- OPERACIONES ---
            { category: "Operaciones", icon: "∪", question: "Probabilidad de la Unión ($A \\cup B$)", answer: "$$P(A \\cup B) = P(A) + P(B) - P(A \\cap B)$$ <small>(Restamos la intersección para no contarla doble)</small>" },
            { category: "Operaciones", icon: "🧠", question: "Leyes de Morgan (1)", answer: "El contrario de la unión es la intersección de contrarios: $$P(\\overline{A \\cup B}) = P(\\bar{A} \\cap \\bar{B})$$" },
            { category: "Operaciones", icon: "🧠", question: "Leyes de Morgan (2)", answer: "El contrario de la intersección es la unión de contrarios: $$P(\\overline{A \\cap B}) = P(\\bar{A} \\cup \\bar{B})$$" },

            { category: "Condicionada", icon: "🤔", question: "Probabilidad Condicionada", answer: "Probabilidad de A sabiendo que ha pasado B: $$P(A|B) = \\frac{P(A \\cap B)}{P(B)}$$" },
            { category: "Condicionada", icon: "✖️", question: "Regla del Producto (Prob. Compuesta)", answer: "Despejando de la condicionada: $$P(A \\cap B) = P(A) \\cdot P(B|A)$$" },

            { category: "Teoremas Clave", icon: "🔙", question: "Teorema de Bayes", answer: "Probabilidad 'hacia atrás' (Causa dado el Efecto): $$P(A_i|B) = \\frac{P(A_i) \\cdot P(B|A_i)}{P(B)}$$" },

        ];

        const grid = document.getElementById('grid');

        function renderCards(filter = 'all') {
            grid.innerHTML = '';
            const filteredData = filter === 'all' ? cardsData : cardsData.filter(card => card.category === filter);

            filteredData.forEach(card => {
                const el = document.createElement('div');
                el.className = 'flashcard';
                el.onclick = function() { this.classList.toggle('flipped'); };
                el.innerHTML = `
                    <div class="flashcard-inner">
                        <div class="flashcard-front">
                            <span class="fc-category">${card.category}</span>
                            <div class="fc-icon">${card.icon}</div>
                            <div class="fc-question">${card.question}</div>
                            <div class="fc-hint">Clic para ver respuesta</div>
                        </div>
                        <div class="flashcard-back">
                            <div class="fc-answer">${card.answer}</div>
                        </div>
                    </div>
                `;
                grid.appendChild(el);
            });

            if (window.MathJax) MathJax.typesetPromise();
        }

        window.filterCards = function(category, btn) {
            document.querySelectorAll('.filter-btn').forEach(b => b.classList.remove('active'));
            btn.classList.add('active');
            renderCards(category);
        };

        renderCards('all');
    </script>
