                explanation: "El último párrafo de muestreo estratificado afirma: «Una vez que se definen los estratos, se procede a extraer una muestra aleatoria simple de cada uno de ellos»."
            },
            {
                question: "Según el texto introductorio, si extraemos una mala elección de muestra (Apunte 2), ¿qué consecuencia directa tiene sobre el estudio?",
                options: [
                    "El tiempo de desplazamiento del encuestador aumentará drásticamente.",
                    "Nos obligará a utilizar obligatoriamente métodos por conglomerados.",
                    "Puede conducir a resultados erróneos o inexactos.",
                    "La población total N cambiará automáticamente sus parámetros estadísticos."
                ],
                correct: 2,
                explanation: "El Apunte 2 de los márgenes indica literalmente: «Una mala elección de la muestra puede conducir a resultados erróneos o inexactos»."
            }
        ];

        // ESTADO DE LA APLICACIÓN
        let currentQuestionIndex = 0;
        let score = 0;
        let hasAnswered = false;

        // ELEMENTOS DEL DOM
        const screens = {
            start: document.getElementById('start-screen'),
            quiz: document.getElementById('quiz-screen'),
            results: document.getElementById('results-screen')
        };
        
        const ui = {
            qCounter: document.getElementById('q-counter'),
            pBar: document.getElementById('p-bar'),
            scoreCounter: document.getElementById('score-counter'),
            qText: document.getElementById('question-text'),
            optionsBox: document.getElementById('options-container'),
            feedback: document.getElementById('feedback'),
            feedbackTitle: document.getElementById('feedback-title'),
            feedbackText: document.getElementById('feedback-text'),
            btnNext: document.getElementById('btn-next')
        };

        const letters = ['A', 'B', 'C', 'D'];

        // LÓGICA DE CONTROL
        function startQuiz() {
            screens.start.style.display = 'none';
            screens.quiz.style.display = 'block';
            renderQuestion();
        }

        function renderQuestion() {
            const q = questions[currentQuestionIndex];
            hasAnswered = false;

            // Actualizar UI
            ui.qCounter.textContent = `PREGUNTA ${currentQuestionIndex + 1} DE ${questions.length}`;
            ui.pBar.style.width = `${((currentQuestionIndex) / questions.length) * 100}%`;
            ui.scoreCounter.textContent = `Aciertos: ${score}`;
            
            ui.qText.innerHTML = q.question;
            ui.feedback.className = 'feedback-box'; // reset classes
            ui.btnNext.style.display = 'none';
            
            // Generar Opciones
            ui.optionsBox.innerHTML = '';
            q.options.forEach((optText, index) => {
                const btn = document.createElement('button');
                btn.className = 'option-btn';
                btn.onclick = () => selectOption(index);
                
                btn.innerHTML = `
                    <span class="option-letter">${letters[index]}</span>
                    <span>${optText}</span>
                `;
                ui.optionsBox.appendChild(btn);
            });

            // Refrescar MathJax por si hay LaTeX
            if (window.MathJax) MathJax.typesetPromise([ui.qText, ui.optionsBox]);
        }

        function selectOption(selectedIndex) {
            if (hasAnswered) return;
            hasAnswered = true;

            const q = questions[currentQuestionIndex];
            const buttons = ui.optionsBox.querySelectorAll('.option-btn');
            
            // Evaluar
            const isCorrect = (selectedIndex === q.correct);
            if (isCorrect) score++;

            // Estilos a los botones
            buttons.forEach((btn, index) => {
                btn.disabled = true;
                if (index === q.correct) {
                    btn.classList.add('correct');
                } else if (index === selectedIndex) {
                    btn.classList.add('incorrect');
                }
            });

            // Mostrar Feedback
            ui.feedback.classList.add('show', isCorrect ? 'correct' : 'incorrect');
            ui.feedbackTitle.textContent = isCorrect ? "¡Correcto!" : "Respuesta Incorrecta";
            ui.feedbackText.innerHTML = q.explanation;
            
            // Actualizar score counter en vivo
            ui.scoreCounter.textContent = `Aciertos: ${score}`;

            // Botón siguiente
            ui.btnNext.textContent = (currentQuestionIndex === questions.length - 1) ? "Ver Resultados Finales ➔" : "Siguiente Pregunta ➔";
            ui.btnNext.style.display = 'block';
            
            if (window.MathJax) MathJax.typesetPromise([ui.feedbackText]);
        }

        function nextQuestion() {
            currentQuestionIndex++;
            if (currentQuestionIndex < questions.length) {
                renderQuestion();
            } else {
                showResults();
            }
        }

        function showResults() {
            screens.quiz.style.display = 'none';
            screens.results.style.display = 'block';
            
            document.getElementById('final-score').textContent = `${score}/${questions.length}`;
            
            const msgTitle = document.getElementById('final-message');
            const msgSub = document.getElementById('final-submessage');
            
            const percentage = (score / questions.length) * 100;
            
            if (percentage === 100) {
                msgTitle.textContent = "¡Excelente! Nivel Experto";
                msgTitle.style.color = "var(--primary)";
                msgSub.textContent = "Dominas a la perfección la teoría de muestreo y las técnicas probabilísticas.";
            } else if (percentage >= 70) {
                msgTitle.textContent = "¡Buen trabajo! Nivel Avanzado";
                msgTitle.style.color = "var(--primary-dark)";
                msgSub.textContent = "Has demostrado un conocimiento sólido de los apuntes, aunque hay pequeños detalles por pulir.";
            } else if (percentage >= 50) {
                msgTitle.textContent = "Aprobado. Nivel Medio";
                msgTitle.style.color = "var(--accent)";
                msgSub.textContent = "Conoces los conceptos básicos, pero te recomendamos repasar las diferencias entre los métodos de muestreo.";
            } else {
                msgTitle.textContent = "Necesitas repasar la teoría";
                msgTitle.style.color = "#ef4444";
                msgSub.textContent = "Te sugerimos releer detenidamente la diferencia entre población y muestra y los métodos probabilísticos y no probabilísticos.";
            }
        }
    </script>
