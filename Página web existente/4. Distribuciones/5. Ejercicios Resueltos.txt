        function renderExercise(exercise) {
            // Reiniciar animación css pura
            container.classList.remove('fade-in');
            void container.offsetWidth; // Trigger reflow
            container.classList.add('fade-in');
            container.classList.remove('hidden');

            exTitle.innerHTML = exercise.title;
            exStatement.innerHTML = exercise.statement;
            exSteps.innerHTML = '';

            exercise.steps.forEach((step, index) => {
                // AQUÍ USAMOS LAS NUEVAS CLASES CSS EN LUGAR DE TAILWIND
                const stepHTML = `
                    <div class="step-card">
                        <div class="step-header">
                            <h4 class="step-title">${step.title}</h4>
                            <span class="step-badge">Paso ${index + 1}</span>
                        </div>
                        <div class="step-content">
                            ${step.content}
                        </div>
                    </div>
                `;
                exSteps.insertAdjacentHTML('beforeend', stepHTML);
            });

            if (exercise.conclusion) {
                exConclusion.innerHTML = `
                    <span class="conclusion-label">Solución Final</span>
                    <div class="conclusion-text">${exercise.conclusion}</div>
                `;
                exConclusion.classList.remove('hidden');
            } else {
                exConclusion.classList.add('hidden');
            }

            // REPROCESAR MATHJAX
            if (window.MathJax) {
                MathJax.typesetPromise([container]).catch(err => console.log(err));
            }
