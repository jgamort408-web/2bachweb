          <p style="margin-bottom:0.8rem;">
            En la normal, la probabilidad se interpreta como <strong>área</strong> bajo la densidad $f(x)$.
            La función acumulada $F(x)$ te da directamente $P(X\le x)$:
          </p>

          <div class="note">
            <strong style="color:var(--primary-dark); display:block; margin-bottom:0.35rem;">
              Densidad vs Distribución
            </strong>
            <p class="mini" style="margin-bottom:0.5rem;">
              <span class="pill mint">Densidad</span> $f(x)$ (forma de campana).
              <br/>
              <span class="pill">Acumulada</span> $F(x)=\int_{-\infty}^{x}f(t)\,dt$ (probabilidad acumulada).
            </p>
            <div class="formula-display">
              $$X\sim N(\mu,\sigma)\qquad
              f(x)=\frac{1}{\sigma\sqrt{2\pi}}e^{-\frac12\left(\frac{x-\mu}{\sigma}\right)^2}
              \qquad
              F(x)=P(X\le x)$$
            </div>
          </div>

          <p class="mini" style="margin-top:0.8rem;">
            Estandarización:
            $$Z=\frac{X-\mu}{\sigma}\sim N(0,1)\quad\Rightarrow\quad P(X\le x)=\Phi\!\left(\frac{x-\mu}{\sigma}\right).$$
          </p>
        </div>

      <h3 style="margin-bottom:0.5rem; text-align:center;">2) Aproximación de la Binomial con la Normal</h3>
      <p class="mini" style="text-align:center; margin-bottom:1.2rem;">
        Si $X\sim \mathrm{Bin}(n,p)$, entonces para $n$ grande la distribución de $X$ se aproxima por una normal:
        $$X \approx N\big(\mu=np,\ \sigma=\sqrt{np(1-p)}\big).$$
        La binomial es <strong>discreta</strong> (barras rojas) y la normal es <strong>continua</strong> (curva verde).
      </p>

      <div class="grid-3" style="margin-bottom:1.5rem;">
        <div class="prop-card">
          <div class="prop-title">Condición típica</div>
          <p class="prop-desc">Para que “se parezcan”:</p>
          <div style="font-family:'Times New Roman'; font-size:1.05rem;">
            $$np\ge 5\quad\text{y}\quad n(1-p)\ge 5$$
          </div>
          <p class="mini" style="margin-top:0.5rem;">
            (Cuanto más grande, mejor.)
          </p>
        </div>

        <div class="prop-card">
          <div class="prop-title">Parámetros</div>
          <p class="prop-desc">Media y desviación para la aproximación:</p>
          <div style="font-family:'Times New Roman'; font-size:1.05rem;">
            $$\mu=np,\qquad \sigma=\sqrt{np(1-p)}$$
          </div>
        </div>

        <div class="prop-card">

          <div style="font-family:'Times New Roman'; font-size:1.05rem;">
            $$P(X=k)\approx P(k-\tfrac12 \le Y \le k+\tfrac12)$$

    // =========================
    // erf + normal CDF/PDF
    // =========================
    function erf(x){
      const sign = x >= 0 ? 1 : -1;
      x = Math.abs(x);
      const a1=0.254829592, a2=-0.284496736, a3=1.421413741, a4=-1.453152027, a5=1.061405429;
      const p=0.3275911;
      const t=1/(1+p*x);
      const y=1-(((((a5*t+a4)*t+a3)*t+a2)*t+a1)*t)*Math.exp(-x*x);
      return sign*y;
    }
    function phi(z){ return 0.5*(1+erf(z/Math.SQRT2)); }
    function normalPDF(x, mu, sigma){
      const z=(x-mu)/sigma;
      return (1/(sigma*Math.sqrt(2*Math.PI)))*Math.exp(-0.5*z*z);
    }
    function normalCDF(x, mu, sigma){
      return phi((x-mu)/sigma);
    }

    // =========================
    // Binomial PMF (estable) por recurrencia
    // =========================
    function binomPMFArray(n, p){
      // Devuelve array pmf[k] = P(X=k) para k=0..n
      const q = 1 - p;
      const pmf = new Array(n+1).fill(0);

      // p0 = q^n
      let pk = Math.pow(q, n);
      pmf[0] = pk;

      // recurrencia: P(k+1)=P(k)*((n-k)/(k+1))*(p/q)
      const ratio = p / q;
      for(let k=0;k<n;k++){
        pk = pk * ((n-k)/(k+1)) * ratio;
        pmf[k+1] = pk;
      }
      return pmf;
    }

    function sumRange(arr, a, b){
      const aa = Math.max(0, Math.min(a,b));
      const bb = Math.min(arr.length-1, Math.max(a,b));
      let s=0;
      for(let k=aa;k<=bb;k++) s += arr[k];
      return s;
    }

    function el(name, attrs={}){
      const node = document.createElementNS("http://www.w3.org/2000/svg", name);
      for(const [k,v] of Object.entries(attrs)) node.setAttribute(k,v);

    function clearSVG(svg){ while(svg.firstChild) svg.removeChild(svg.firstChild); }
    function mapper(xmin, xmax, ymin, ymax, W, H, pad){
      return {
        X: x => pad + (x-xmin)*(W-2*pad)/(xmax-xmin),
        Y: y => (H-pad) - (y-ymin)*(H-2*pad)/(ymax-ymin)
      };
    }
    function pathFromPoints(points){
      if(!points.length) return "";
      let d = `M ${points[0][0].toFixed(2)} ${points[0][1].toFixed(2)}`;
      for(let i=1;i<points.length;i++){
        d += ` L ${points[i][0].toFixed(2)} ${points[i][1].toFixed(2)}`;
      }
      return d;
    }

    // =========================
    // Draw small PDF/CDF (normal)
    // =========================
    function drawPDFSmall(svg, mu, sigma){
      const W=420,H=210,pad=40;
      const xmin = mu - 4*sigma, xmax = mu + 4*sigma;
      const ymax = normalPDF(mu, mu, sigma)*1.2;
      const m = mapper(xmin, xmax, 0, ymax, W, H, pad);
      clearSVG(svg);

      // grid
      for(let i=0;i<=6;i++){
        const x=xmin+i*(xmax-xmin)/6;
        svg.appendChild(el("line",{x1:m.X(x),y1:pad,x2:m.X(x),y2:H-pad,class:"gridline"}));
      }
      svg.appendChild(el("line",{x1:pad,y1:H-pad,x2:W-pad,y2:H-pad,class:"axis"}));
      svg.appendChild(el("line",{x1:pad,y1:pad,x2:pad,y2:H-pad,class:"axis"}));

      // curve
      const N=160, pts=[];
      for(let i=0;i<=N;i++){
        const x=xmin+i*(xmax-xmin)/N;
        pts.push([m.X(x), m.Y(normalPDF(x,mu,sigma))]);
      }
      svg.appendChild(el("path",{d:pathFromPoints(pts), class:"curve-normal"}));

      // mu and +-sigma
      [mu-sigma, mu, mu+sigma].forEach((x,idx)=>{
        svg.appendChild(el("line",{x1:m.X(x),y1:pad,x2:m.X(x),y2:H-pad,class:"dashed"}));
      });
      svg.appendChild(el("text",{x:pad,y:22,class:"lblStrong"})).textContent="Densidad f(x)";
    }

    function drawCDFSmall(svg, mu, sigma){
      const W=420,H=210,pad=40;
      const xmin = mu - 4*sigma, xmax = mu + 4*sigma;
      const m = mapper(xmin, xmax, 0, 1, W, H, pad);
      clearSVG(svg);

      for(let i=0;i<=6;i++){
        const x=xmin+i*(xmax-xmin)/6;
        svg.appendChild(el("line",{x1:m.X(x),y1:pad,x2:m.X(x),y2:H-pad,class:"gridline"}));
      }
      svg.appendChild(el("line",{x1:pad,y1:H-pad,x2:W-pad,y2:H-pad,class:"axis"}));
      svg.appendChild(el("line",{x1:pad,y1:pad,x2:pad,y2:H-pad,class:"axis"}));

      const N=160, pts=[];
      for(let i=0;i<=N;i++){
        const x=xmin+i*(xmax-xmin)/N;
        pts.push([m.X(x), m.Y(normalCDF(x,mu,sigma))]);

      svg.appendChild(el("text",{x:pad,y:22,class:"lblStrong"})).textContent="Acumulada F(x)";
    }

    function drawBinomVsNormal(svg, n, p, mode, K, A, B){
      const W=680,H=320,pad=52;
      clearSVG(svg);

      const pmf = binomPMFArray(n, p);
      const mu = n*p;
      const sigma = Math.sqrt(n*p*(1-p));

      // x-range: 0..n, but pad around mean for nicer look
      const xmin = 0, xmax = n;

      // y-range based on max of pmf and normal pdf near mean (scaled to bars)
      let maxP = 0;
      for(let k=0;k<=n;k++) if(pmf[k]>maxP) maxP=pmf[k];
      // normal pdf is density; to compare with bar heights, multiply by 1 (bin width)
      const maxD = normalPDF(mu, mu, sigma);
      const ymax = Math.max(maxP, maxD) * 1.25;

      const m = mapper(xmin, xmax, 0, ymax, W, H, pad);

      // grid
      for(let i=0;i<=8;i++){
        const x=xmin+i*(xmax-xmin)/8;
        svg.appendChild(el("line",{x1:m.X(x),y1:pad,x2:m.X(x),y2:H-pad,class:"gridline"}));
      }
      for(let j=0;j<=4;j++){
        const y=j*ymax/4;
        svg.appendChild(el("line",{x1:pad,y1:m.Y(y),x2:W-pad,y2:m.Y(y),class:"gridline"}));
      }

      // axes
      svg.appendChild(el("line",{x1:pad,y1:H-pad,x2:W-pad,y2:H-pad,class:"axis"}));
      svg.appendChild(el("line",{x1:pad,y1:pad,x2:pad,y2:H-pad,class:"axis"}));

      // bars (binomial)
      const barW = (W-2*pad) / (xmax-xmin+1);
      for(let k=0;k<=n;k++){
        const xLeft = m.X(k) - barW*0.45;
        const h = (H-pad) - m.Y(pmf[k]);
        svg.appendChild(el("rect",{
          x:xLeft, y: (H-pad)-h, width: barW*0.9, height:h,
          class:"binom-bars"
        }));
      }

      // normal curve (density), draw across 0..n
      const N=320, pts=[];
      for(let i=0;i<=N;i++){
        const x = xmin + i*(xmax-xmin)/N;
        const y = normalPDF(x, mu, sigma); // density
        pts.push([m.X(x), m.Y(y)]);
      }
      svg.appendChild(el("path",{d:pathFromPoints(pts), class:"curve-normal"}));

      // shading for continuity correction
      let x1, x2;
      if(mode===0){
        // P(X<=K) ~ P(Y<=K+0.5): shade from xmin to K+0.5 under normal (conceptually)
        x1 = xmin;
        x2 = Math.min(xmax, K + 0.5);
      }else{
        // interval A..B => [A-0.5, B+0.5]
        const aa = Math.min(A,B), bb = Math.max(A,B);
        x1 = Math.max(xmin, aa - 0.5);
        x2 = Math.min(xmax, bb + 0.5);
      }

      const shadePts = [];
      shadePts.push([m.X(x1), m.Y(0)]);
      const Ns=220;
      for(let i=0;i<=Ns;i++){
        const x = x1 + i*(x2-x1)/Ns;
        shadePts.push([m.X(x), m.Y(normalPDF(x, mu, sigma))]);
      }
      shadePts.push([m.X(x2), m.Y(0)]);
      svg.appendChild(el("path",{d:pathFromPoints(shadePts)+" Z", class:"shade-normal"}));

      // guide lines and labels
      svg.appendChild(el("text",{x:pad,y:24,class:"lblStrong"})).textContent="Binomial (barras) vs Normal (curva) + corrección de continuidad";

      // mu line
      svg.appendChild(el("line",{x1:m.X(mu),y1:pad,x2:m.X(mu),y2:H-pad,class:"dashed"}));
      svg.appendChild(el("text",{x:m.X(mu)-10,y:H-pad+22,class:"lbl"})).textContent="μ=np";

      // draw x2 marker
      svg.appendChild(el("line",{x1:m.X(x2),y1:pad,x2:m.X(x2),y2:H-pad,class:"dashed"}));
      svg.appendChild(el("text",{x:m.X(x2)-20,y:pad+16,class:"lbl"})).textContent=(mode===0? "K+0.5" : "B+0.5");

      if(mode===1){
        svg.appendChild(el("line",{x1:m.X(x1),y1:pad,x2:m.X(x1),y2:H-pad,class:"dashed"}));
        svg.appendChild(el("text",{x:m.X(x1)-20,y:pad+16,class:"lbl"})).textContent="A-0.5";
      }
    }

    // =========================
    // Draw CDF comparison for cumulative
    // =========================
    function drawApproxCDF(svg, n, p, K){
      const W=680,H=320,pad=52;
      clearSVG(svg);

      const pmf = binomPMFArray(n, p);
      const mu = n*p;
      const sigma = Math.sqrt(n*p*(1-p));

      const xmin=0, xmax=n;
      const m = mapper(xmin, xmax, 0, 1, W, H, pad);

      // grid
      for(let i=0;i<=8;i++){
        const x=xmin+i*(xmax-xmin)/8;
        svg.appendChild(el("line",{x1:m.X(x),y1:pad,x2:m.X(x),y2:H-pad,class:"gridline"}));
      }
      for(let j=0;j<=5;j++){
        const y=j/5;
        svg.appendChild(el("line",{x1:pad,y1:m.Y(y),x2:W-pad,y2:m.Y(y),class:"gridline"}));
        svg.appendChild(el("text",{x:14,y:m.Y(y)+4,class:"lbl"})).textContent=y.toFixed(1);
      }

      // axes
      svg.appendChild(el("line",{x1:pad,y1:H-pad,x2:W-pad,y2:H-pad,class:"axis"}));
      svg.appendChild(el("line",{x1:pad,y1:pad,x2:pad,y2:H-pad,class:"axis"}));

      // Binomial CDF (step)
      const stepPts = [];
      let acc = 0;
      stepPts.push([m.X(0), m.Y(0)]);
      for(let k=0;k<=n;k++){
        const x = k;
        const y0 = acc;
        acc += pmf[k];
        const y1 = acc;

        // horizontal at y0 until x
        stepPts.push([m.X(x), m.Y(y0)]);
        // vertical jump
        stepPts.push([m.X(x), m.Y(y1)]);
      }
      svg.appendChild(el("path",{d:pathFromPoints(stepPts), fill:"none", stroke:"rgba(180,83,98,0.85)", "stroke-width":"2.4"}));

      // Normal CDF curve
      const N=320, pts=[];
      for(let i=0;i<=N;i++){
        const x=xmin+i*(xmax-xmin)/N;
        pts.push([m.X(x), m.Y(normalCDF(x, mu, sigma))]);
      }
      svg.appendChild(el("path",{d:pathFromPoints(pts), class:"curve-cdf"}));

      // marker at K and K+0.5
      const xMark = Math.min(xmax, K);
      const xCont = Math.min(xmax, K+0.5);
      const Fbin = (function(){
        let s=0;
        for(let i=0;i<=Math.min(n,K);i++) s += pmf[i];
        return s;
      })();
      const Fnorm = normalCDF(xCont, mu, sigma);

      // shade conceptual until K+0.5
      const shadePts = [
        [m.X(xmin), m.Y(0)],
        [m.X(xmin), m.Y(1)],
        [m.X(xCont), m.Y(1)],
        [m.X(xCont), m.Y(0)]
      ];
      svg.appendChild(el("path",{d:pathFromPoints(shadePts)+" Z", class:"shade-cdf"}));

      // guide lines
      svg.appendChild(el("line",{x1:m.X(xCont),y1:H-pad,x2:m.X(xCont),y2:m.Y(Fnorm),class:"dashed"}));
      svg.appendChild(el("line",{x1:pad,y1:m.Y(Fnorm),x2:m.X(xCont),y2:m.Y(Fnorm),class:"dashed"}));

      // points
      svg.appendChild(el("circle",{cx:m.X(xCont),cy:m.Y(Fnorm),r:4.5,fill:"rgba(15,23,42,0.85)"}));

      // labels
      svg.appendChild(el("text",{x:pad,y:24,class:"lblStrong"})).textContent="Acumulada: Binomial (escalón) vs Normal (S) con corrección K+0.5";
      svg.appendChild(el("text",{x:m.X(xMark)-6,y:H-pad+22,class:"lbl"})).textContent="K";
      svg.appendChild(el("text",{x:m.X(xCont)-20,y:pad+16,class:"lbl"})).textContent="K+0.5";

      svg.appendChild(el("text",{x:pad+6,y:m.Y(Fnorm)-8,class:"lbl"})).textContent=`Normal: F(K+0.5) ≈ ${Fnorm.toFixed(4)}`;
      svg.appendChild(el("text",{x:pad+6,y:m.Y(Fbin)+14,class:"lbl"})).textContent=`Binomial: P(X≤K) = ${Fbin.toFixed(4)}`;
    }

    // =========================
    // UI wiring
    // =========================
    const $ = (id)=>document.getElementById(id);

    const inputs = {
      n: $("n"),
      p: $("p"),
      K: $("K"),
      mode: $("mode"),
      A: $("A"),
      B: $("B")
    };

    const labels = {
      nVal: $("nVal"),
      pVal: $("pVal"),
      KVal: $("KVal"),
      modeVal: $("modeVal"),
      AVal: $("AVal"),
      BVal: $("BVal"),
      readouts: $("readouts")
    };

    const svgPDFSmall = $("svgPDFSmall");
    const svgCDFSmall = $("svgCDFSmall");
    const svgBinomVsNormal = $("svgBinomVsNormal");
    const svgApproxCDF = $("svgApproxCDF");

    function clamp(v, lo, hi){ return Math.max(lo, Math.min(hi, v)); }

    function render(){
      const n = parseInt(inputs.n.value, 10);
      const p = parseFloat(inputs.p.value);
      const K = parseInt(inputs.K.value, 10);
      const mode = parseInt(inputs.mode.value, 10);
      let A = parseInt(inputs.A.value, 10);
      let B = parseInt(inputs.B.value, 10);

      // keep A,B,K within [0,n]
      const KK = clamp(K, 0, n);
      A = clamp(A, 0, n);
      B = clamp(B, 0, n);

      // update max for sliders dependent on n
      inputs.K.max = n;
      inputs.A.max = n;
      inputs.B.max = n;

      labels.nVal.textContent = n;
      labels.pVal.textContent = p.toFixed(2);
      labels.KVal.textContent = KK;
      labels.AVal.textContent = A;
      labels.BVal.textContent = B;
      labels.modeVal.textContent = mode===0 ? "Acumulada" : "Intervalo";

      const mu = n*p;
      const sigma = Math.sqrt(n*p*(1-p));

      const pmf = binomPMFArray(n, p);
      const q = 1-p;

      const cond1 = n*p;
      const cond2 = n*q;

      // exact binomial probs
      let Pbin, Pnorm;
      if(mode===0){
        // P(X <= K)
        Pbin = sumRange(pmf, 0, KK);
        // continuity: Y <= K+0.5
        Pnorm = normalCDF(KK + 0.5, mu, sigma);
      }else{
        const aa = Math.min(A,B);
        const bb = Math.max(A,B);
        Pbin = sumRange(pmf, aa, bb);
        Pnorm = normalCDF(bb + 0.5, mu, sigma) - normalCDF(aa - 0.5, mu, sigma);
      }

      const ok = (cond1>=5 && cond2>=5);
      const zK = ( (KK + 0.5) - mu) / sigma;

      labels.readouts.innerHTML = `
        <div><strong>Parámetros de la normal aproximada:</strong> $\\mu=np=${mu.toFixed(3)}$, $\\sigma=\\sqrt{np(1-p)}=${sigma.toFixed(3)}$</div>
        <div><strong>Condición:</strong> $np=${cond1.toFixed(2)}$, $n(1-p)=${cond2.toFixed(2)}$ → ${ok ? "<strong style='color:var(--mint-dark)'>OK</strong>" : "<strong style='color:var(--primary-dark)'>mejorable</strong>"} </div>
        <div style="margin-top:0.4rem;">
          <span class="pill">Binomial</span>
          ${mode===0
            ? `$P(X\\le K)=${Pbin.toFixed(5)}$`
            : `$P(${Math.min(A,B)}\\le X\\le ${Math.max(A,B)})=${Pbin.toFixed(5)}$`
          }
        </div>
        <div>
          <span class="pill mint">Normal (corr.)</span>
          ${mode===0
            ? `$P(Y\\le K+0.5)=${Pnorm.toFixed(5)}$ &nbsp; (aquí $z=\\frac{K+0.5-\\mu}{\\sigma}=${zK.toFixed(3)}$)`
            : `$P(A-0.5\\le Y\\le B+0.5)=${Pnorm.toFixed(5)}$`
          }
        </div>
        <div class="mini" style="margin-top:0.4rem;">
          Diferencia (error absoluto): $\\lvert ${Pbin.toFixed(5)} - ${Pnorm.toFixed(5)}\\rvert = ${Math.abs(Pbin-Pnorm).toFixed(5)}$.
        </div>
      `;

      // redraw
      drawPDFSmall(svgPDFSmall, 0, 1);
      drawCDFSmall(svgCDFSmall, 0, 1);
      drawBinomVsNormal(svgBinomVsNormal, n, p, mode, KK, A, B);
      drawApproxCDF(svgApproxCDF, n, p, KK);

      if(window.MathJax && MathJax.typesetPromise){
        MathJax.typesetPromise([labels.readouts]);
      }
    }

    Object.values(inputs).forEach(inp => inp.addEventListener("input", render));
    render();
  </script>
