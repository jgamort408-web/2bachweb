      <p class="subtitle">
        El modelo continuo más importante: <strong>la Campana de Gauss</strong>.  
        Aquí verás (de verdad) cómo la probabilidad se convierte en <strong>área</strong> y cómo funciona la <strong>función de distribución</strong> $F(x)$.
      </p>

    <!-- 1) Definición + PDF vs CDF -->
    <section class="card">
      <div class="definition-grid">
        <div>
          <h3 style="margin-bottom: 0.8rem;">1) ¿Qué significa “normal”?</h3>
          <p style="margin-bottom: 1rem;">
            La normal sirve para modelizar variables <strong>continuas</strong> (estaturas, pesos, errores de medida, tiempos…).
            En variables continuas, una “probabilidad puntual” no tiene sentido:
            $$P(X = 3.00)=0$$
            Lo que sí tiene sentido es la probabilidad de caer en un <strong>intervalo</strong>, por ejemplo $P(2.9 \le X \le 3.1)$.
          </p>

          <div class="note">
            <strong style="color: var(--primary-dark); display:block; margin-bottom: 0.4rem;">
              Dos funciones distintas (¡y se confunden mucho!)
            </strong>
            <p class="mini" style="margin-bottom: 0.6rem;">
              <span class="pill">Densidad</span> $f(x)$ describe la forma de la campana (no da probabilidades directamente).
              <br/>
              <span class="pill">Distribución</span> $F(x)$ acumula el área desde $-\infty$ hasta $x$ (eso sí es probabilidad).
            </p>

            <div class="formula-display">
              $$X \sim N(\mu,\sigma)\qquad
              f(x)=\frac{1}{\sigma\sqrt{2\pi}}\,e^{-\frac12\left(\frac{x-\mu}{\sigma}\right)^2}\qquad
              F(x)=P(X\le x)=\int_{-\infty}^{x} f(t)\,dt$$
            </div>

            <p class="mini" style="margin-top: 0.7rem;">
              Idea clave: <strong>$F(x)$ es una función creciente</strong> que va de 0 a 1.  
              En cambio, <strong>$f(x)$</strong> sube y baja, y su área total vale 1.
            </p>
          </div>

          <div style="margin-top: 1.1rem;">
            <h4 style="margin-bottom: 0.4rem;">Parámetros</h4>
            <ul style="padding-left: 1.2rem; color: var(--text-main);">
              <li><strong>Media</strong> $\mu$: centro de la campana (simetría).</li>
              <li><strong>Desviación típica</strong> $\sigma$: dispersión (más grande ⇒ campana más ancha y bajita).</li>
              <li>Puntos de inflexión de $f(x)$ en <strong>$\mu\pm\sigma$</strong>.</li>
            </ul>
          </div>
        </div>

        <div>
          <h4 style="margin-bottom: 0.7rem; font-size: 0.95rem; text-transform: uppercase; color: var(--text-light); text-align:center;">
            PDF y CDF (precisas)

          </div>
          <p class="mini" style="text-align:center; margin-top: 0.6rem;">
            Arriba: densidad $f(x)$. La probabilidad se mide como <strong>área</strong>.

    <h3 style="margin-bottom: 1rem; display:flex; align-items:center; gap:0.5rem;">
      <span style="color: var(--primary);">#</span> 2) La probabilidad es área (y se calcula con $F$)
    </h3>

    <div class="grid-3">
      <div class="prop-card">
        <div class="prop-title">Área total = 1</div>
        <p class="prop-desc">Toda la curva suma el 100% de probabilidad.</p>
        <div style="font-family: 'Times New Roman'; font-size: 1.1rem;">
          $$\int_{-\infty}^{+\infty} f(x)\,dx = 1$$
        </div>
      </div>

      <div class="prop-card">
        <div class="prop-title">Intervalo $(a,b)$</div>
        <p class="prop-desc">La probabilidad entre dos valores es un área:</p>
        <div style="font-family: 'Times New Roman'; font-size: 1.1rem;">
          $$P(a\le X\le b)=$$$$\int_{a}^{b}f(x)\,dx = F(b)-F(a)$$
        </div>
      </div>

      <div class="prop-card">
        <div class="prop-title">Acumulada $P(X\le k)$</div>
        <p class="prop-desc">Es el área desde $-\infty$ hasta $k$:</p>
        <div style="font-family: 'Times New Roman'; font-size: 1.1rem;">
          $$P(X\le k)=F(k)$$
        </div>
      </div>
    </div>

    <section class="card">
      <h3 style="margin-bottom: 0.8rem;">3) Estandarización y tablas (lo que se hace en exámenes)</h3>
      <p style="margin-bottom: 0.8rem;">
        En Bachillerato, lo habitual es convertir cualquier normal en la <strong>normal estándar</strong>:
        $$Z=\frac{X-\mu}{\sigma}\qquad\Rightarrow\qquad Z\sim N(0,1)$$
      </p>

      <div class="note">
        <strong style="color: var(--primary-dark); display:block; margin-bottom: 0.35rem;">
          Receta rápida
        </strong>
        <ol style="padding-left: 1.2rem;">
          <li>Transforma a puntuación típica: $z=\dfrac{x-\mu}{\sigma}$.</li>
          <li>Busca $\Phi(z)=P(Z\le z)$ en la tabla (o calcula con calculadora/software).</li>
          <li>Vuelve a la probabilidad pedida:
            <ul style="padding-left: 1.2rem;">
              <li>$P(X\le x)=\Phi\!\left(\dfrac{x-\mu}{\sigma}\right)$</li>
              <li>$P(a\le X\le b)=\Phi\!\left(\dfrac{b-\mu}{\sigma}\right)-\Phi\!\left(\dfrac{a-\mu}{\sigma}\right)$</li>
            </ul>
          </li>
        </ol>
      </div>

      <p class="mini" style="margin-top: 0.8rem;">
        Nota práctica: por simetría, $\Phi(-z)=1-\Phi(z)$.
      </p>
    </section>

    // =========================
    // Matemáticas: PDF y CDF normal
    // =========================

    // Aproximación de erf(x) (Abramowitz & Stegun, forma clásica)
    function erf(x) {
      const sign = x >= 0 ? 1 : -1;
      x = Math.abs(x);

      const a1 = 0.254829592;
      const a2 = -0.284496736;
      const a3 = 1.421413741;
      const a4 = -1.453152027;
      const a5 = 1.061405429;
      const p  = 0.3275911;

      const t = 1.0 / (1.0 + p * x);
      const y = 1.0 - (((((a5 * t + a4) * t) + a3) * t + a2) * t + a1) * t * Math.exp(-x * x);
      return sign * y;
    }

    function phi(z) { // CDF estándar
      return 0.5 * (1 + erf(z / Math.SQRT2));
    }

    function normalPDF(x, mu, sigma) {
      const z = (x - mu) / sigma;
      return (1 / (sigma * Math.sqrt(2 * Math.PI))) * Math.exp(-0.5 * z * z);
    }

    function normalCDF(x, mu, sigma) {
      return phi((x - mu) / sigma);
    }

    // =========================
    // SVG helpers
    // =========================
    function el(name, attrs = {}) {
      const node = document.createElementNS("http://www.w3.org/2000/svg", name);
      for (const [k, v] of Object.entries(attrs)) node.setAttribute(k, v);
      return node;
    }

    function clearSVG(svg) {
      while (svg.firstChild) svg.removeChild(svg.firstChild);
    }

    // Map from data coords to SVG coords
    function mapper(xmin, xmax, ymin, ymax, W, H, pad) {
      return {
        X: x => pad + (x - xmin) * (W - 2*pad) / (xmax - xmin),
        Y: y => (H - pad) - (y - ymin) * (H - 2*pad) / (ymax - ymin)
      };
    }

    function pathFromPoints(points) {
      if (!points.length) return "";
      let d = `M ${points[0][0].toFixed(2)} ${points[0][1].toFixed(2)}`;
      for (let i = 1; i < points.length; i++) {
        d += ` L ${points[i][0].toFixed(2)} ${points[i][1].toFixed(2)}`;
      }
      return d;
    }

    // =========================
    // Dibujo de PDF con sombreado
    // =========================
    function drawPDF(svg, mu, sigma, a, b, k, opts = {}) {
      const W = 680, H = 320, pad = 48;
      const isSmall = !!opts.small;

      const xmin = mu - 4 * sigma;
      const xmax = mu + 4 * sigma;

      // encontrar ymax (máximo de la densidad en x=mu)
      const ymax = normalPDF(mu, mu, sigma) * 1.15;

      const m = mapper(xmin, xmax, 0, ymax, W, H, pad);

      clearSVG(svg);

      // fondo y rejilla suave
      for (let i = 0; i <= 8; i++) {
        const x = xmin + i * (xmax - xmin) / 8;
        svg.appendChild(el("line", { x1: m.X(x), y1: pad, x2: m.X(x), y2: H - pad, class: "gridline" }));
      }
      for (let j = 0; j <= 4; j++) {
        const y = j * ymax / 4;
        svg.appendChild(el("line", { x1: pad, y1: m.Y(y), x2: W - pad, y2: m.Y(y), class: "gridline" }));
      }

      // ejes
      svg.appendChild(el("line", { x1: pad, y1: H - pad, x2: W - pad, y2: H - pad, class: "axis" }));
      svg.appendChild(el("line", { x1: pad, y1: pad, x2: pad, y2: H - pad, class: "axis" }));

      // curva
      const N = isSmall ? 160 : 260;
      const pts = [];
      for (let i = 0; i <= N; i++) {
        const x = xmin + i * (xmax - xmin) / N;
        const y = normalPDF(x, mu, sigma);
        pts.push([m.X(x), m.Y(y)]);
      }
      svg.appendChild(el("path", { d: pathFromPoints(pts), class: isSmall ? "curve2" : "curve" }));

      // sombreado intervalo [a,b]
      const aa = Math.min(a, b), bb = Math.max(a, b);
      const shadePts = [];
      shadePts.push([m.X(aa), m.Y(0)]);
      for (let i = 0; i <= N; i++) {
        const x = aa + i * (bb - aa) / N;
        const y = normalPDF(x, mu, sigma);
        shadePts.push([m.X(x), m.Y(y)]);
      }
      shadePts.push([m.X(bb), m.Y(0)]);

      svg.appendChild(el("path", { d: pathFromPoints(shadePts) + " Z", class: "shade" }));

      // línea mu, mu±sigma
      const marks = [
        { x: mu, label: "μ" },
        { x: mu - sigma, label: "μ-σ" },
        { x: mu + sigma, label: "μ+σ" },
      ];

      marks.forEach(mark => {
        svg.appendChild(el("line", { x1: m.X(mark.x), y1: pad, x2: m.X(mark.x), y2: H - pad, class: "dashed" }));
        svg.appendChild(el("text", { x: m.X(mark.x) - 10, y: H - pad + 22, class: "lbl" })).textContent = mark.label;
      });

      // línea vertical en k
      svg.appendChild(el("line", { x1: m.X(k), y1: pad, x2: m.X(k), y2: H - pad, class: "dashed" }));
      svg.appendChild(el("text", { x: m.X(k) - 8, y: H - pad + 22, class: "lbl" })).textContent = "k";

      // título pequeño
      svg.appendChild(el("text", { x: pad, y: 26, class: "lblStrong" })).textContent = isSmall ? "Densidad f(x)" : "PDF: densidad f(x) y área P(a≤X≤b)";
    }

    // =========================
    // Dibujo de CDF con lectura F(k)
    // =========================
    function drawCDF(svg, mu, sigma, a, b, k, opts = {}) {
      const W = 680, H = 320, pad = 48;
      const isSmall = !!opts.small;

      const xmin = mu - 4 * sigma;
      const xmax = mu + 4 * sigma;

      const m = mapper(xmin, xmax, 0, 1, W, H, pad);

      clearSVG(svg);

      // rejilla
      for (let i = 0; i <= 8; i++) {
        const x = xmin + i * (xmax - xmin) / 8;
        svg.appendChild(el("line", { x1: m.X(x), y1: pad, x2: m.X(x), y2: H - pad, class: "gridline" }));
      }
      for (let j = 0; j <= 5; j++) {
        const y = j / 5;
        svg.appendChild(el("line", { x1: pad, y1: m.Y(y), x2: W - pad, y2: m.Y(y), class: "gridline" }));
        // labels 0..1
        if (!isSmall) {
          svg.appendChild(el("text", { x: 14, y: m.Y(y)+4, class: "lbl" })).textContent = y.toFixed(1);
        }
      }

      // ejes
      svg.appendChild(el("line", { x1: pad, y1: H - pad, x2: W - pad, y2: H - pad, class: "axis" }));
      svg.appendChild(el("line", { x1: pad, y1: pad, x2: pad, y2: H - pad, class: "axis" }));

      // curva F
      const N = isSmall ? 160 : 260;
      const pts = [];
      for (let i = 0; i <= N; i++) {
        const x = xmin + i * (xmax - xmin) / N;
        const y = normalCDF(x, mu, sigma);
        pts.push([m.X(x), m.Y(y)]);
      }
      svg.appendChild(el("path", { d: pathFromPoints(pts), class: isSmall ? "curve2" : "curve" }));

      // sombreado: recta vertical en k y horizontal en F(k)
      const Fk = normalCDF(k, mu, sigma);

      // área conceptual (banda) hasta k (opcional, ligera)
      const shadePts = [];
      shadePts.push([m.X(xmin), m.Y(0)]);
      shadePts.push([m.X(xmin), m.Y(1)]);
      shadePts.push([m.X(k), m.Y(1)]);
      shadePts.push([m.X(k), m.Y(0)]);
      svg.appendChild(el("path", { d: pathFromPoints(shadePts) + " Z", class: "shade2" }));

      // guía k -> F(k)
      svg.appendChild(el("line", { x1: m.X(k), y1: H - pad, x2: m.X(k), y2: m.Y(Fk), class: "dashed" }));
      svg.appendChild(el("line", { x1: pad, y1: m.Y(Fk), x2: m.X(k), y2: m.Y(Fk), class: "dashed" }));

      // punto en la curva
      svg.appendChild(el("circle", { cx: m.X(k), cy: m.Y(Fk), r: 4.5, fill: "#be123c" }));

      // etiquetas
      svg.appendChild(el("text", { x: m.X(k) - 8, y: H - pad + 22, class: "lbl" })).textContent = "k";
      if (!isSmall) {
        svg.appendChild(el("text", { x: pad + 6, y: m.Y(Fk) - 8, class: "lbl" }))
          .textContent = `F(k) = ${Fk.toFixed(4)}`;
      }

      // título pequeño
      svg.appendChild(el("text", { x: pad, y: 26, class: "lblStrong" })).textContent = isSmall ? "Acumulada F(x)" : "CDF: F(x)=P(X≤x) y lectura de F(k)";
    }

    // =========================
    // UI logic
    // =========================
    const $ = (id) => document.getElementById(id);

    const inputs = {
      mu: $("mu"),
      sigma: $("sigma"),
      a: $("a"),
      b: $("b"),
      k: $("k"),
    };

    const labels = {
      muVal: $("muVal"),
      sigmaVal: $("sigmaVal"),
      aVal: $("aVal"),
      bVal: $("bVal"),
      kVal: $("kVal"),
      readouts: $("readouts"),
    };

    const svgPDF = $("svgPDF");
    const svgCDF = $("svgCDF");
    const svgPDFSmall = $("svgPDFSmall");
    const svgCDFSmall = $("svgCDFSmall");

    function clamp(v, lo, hi) { return Math.max(lo, Math.min(hi, v)); }

    function render() {
      const mu = parseFloat(inputs.mu.value);
      const sigma = parseFloat(inputs.sigma.value);

      // Convertimos a, b, k (en escala de "x") pero garantizando que se vean bien respecto a mu y sigma
      let a = parseFloat(inputs.a.value);
      let b = parseFloat(inputs.b.value);
      let k = parseFloat(inputs.k.value);

      // Para que el alumno no se pierda: hacemos que a,b,k estén en torno a mu con escala sigma (opcional).
      // Aquí los dejamos como valores absolutos, pero los límites de dibujo dependen de mu y sigma.
      // Ajuste suave si sigma cambia mucho:
      const xmin = mu - 4*sigma;
      const xmax = mu + 4*sigma;
      a = clamp(a, xmin, xmax);
      b = clamp(b, xmin, xmax);
      k = clamp(k, xmin, xmax);

      labels.muVal.textContent = mu.toFixed(1);
      labels.sigmaVal.textContent = sigma.toFixed(1);
      labels.aVal.textContent = a.toFixed(1);
      labels.bVal.textContent = b.toFixed(1);
      labels.kVal.textContent = k.toFixed(1);

      // Probabilidades
      const aa = Math.min(a, b), bb = Math.max(a, b);
      const Pa = normalCDF(aa, mu, sigma);
      const Pb = normalCDF(bb, mu, sigma);
      const Pab = Pb - Pa;

      const Fk = normalCDF(k, mu, sigma);

      const zk = (k - mu) / sigma;
      const za = (aa - mu) / sigma;
      const zb = (bb - mu) / sigma;

      labels.readouts.innerHTML = `
        <div><strong>Acumulada:</strong> $F(k)=P(X\\le k)=${Fk.toFixed(4)}$</div>
        <div><strong>Intervalo:</strong> $P(${aa.toFixed(1)}\\le X\\le ${bb.toFixed(1)})=F(${bb.toFixed(1)})-F(${aa.toFixed(1)})=${Pab.toFixed(4)}$</div>
        <div class="mini" style="margin-top:0.35rem;">
          Estandarizando:
          $z_k=\\dfrac{k-\\mu}{\\sigma}=${zk.toFixed(3)}$,
          $z_a=${za.toFixed(3)}$,
          $z_b=${zb.toFixed(3)}$.
          <br/>
          Entonces: $F(k)=\\Phi(z_k)$ y $P(a\\le X\\le b)=\\Phi(z_b)-\\Phi(z_a)$.
        </div>
      `;

      // Re-render MathJax (cuando cambiamos el HTML con LaTeX)
      if (window.MathJax && MathJax.typesetPromise) MathJax.typesetPromise([labels.readouts]);

      // Dibujo grande
      drawPDF(svgPDF, mu, sigma, a, b, k);
      drawCDF(svgCDF, mu, sigma, a, b, k);

      // Dibujo pequeño (estético) con mu/sigma actuales y un sombreado simple
      drawPDF(svgPDFSmall, mu, sigma, mu - sigma, mu + sigma, mu, { small: true });
      drawCDF(svgCDFSmall, mu, sigma, mu - sigma, mu + sigma, mu + 0.8*sigma, { small: true });
    }

    Object.values(inputs).forEach(inp => inp.addEventListener("input", render));
    render();
  </script>
