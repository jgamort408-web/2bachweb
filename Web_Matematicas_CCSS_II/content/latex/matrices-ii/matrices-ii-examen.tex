\section*{Ejercicio 1 \hfill 5 puntos}
Se consideran las matrices
\[
A=\begin{pmatrix}1&1&3\\1&2&-1\\1&-1&-1\end{pmatrix},\qquad
B=\begin{pmatrix}2\\3\\0\end{pmatrix}.
\]
\begin{enumerate}[label=(\alph*)]
\item (3 puntos) Justifica que $A$ tiene inversa y calcula $A^{-1}$.
\item (2 puntos) Calcula, si existe, la matriz $X$ que satisface $AX=B$.
\end{enumerate}
\section*{Ejercicio 2 \hfill 2,5 puntos}
En este ejercicio se consideran las matrices
\[
A=\begin{pmatrix}1&-1\\0&1\end{pmatrix},\quad
B=\begin{pmatrix}2&0\\-1&2\end{pmatrix},\quad
C=\begin{pmatrix}0\\1\end{pmatrix},\quad
D=\begin{pmatrix}-1&2\end{pmatrix}.
\]
\textbf{Efectúa cuatro de las cinco operaciones} cuando sean posibles; si alguna no es posible, justifica por qué:
\[
AC+BD^t,\qquad A+BC,\qquad B^2+CD,\qquad |5A|,\qquad |C|.
\]
\section*{Ejercicio 3 \hfill 2,5 puntos}
Determina $a$ y $b$ para que se cumpla
\[
\begin{pmatrix}2&-1\\3&-1\end{pmatrix}\begin{pmatrix}a\\-b\end{pmatrix}
=\begin{pmatrix}1&2\\b&-1\end{pmatrix}\begin{pmatrix}3\\0\end{pmatrix}.
\]
