\cabecera{2}{Representación de funciones}{Lecciones 2.1--2.9: esquema general, polinómicas, racionales, exponenciales, logarítmicas, funciones a trozos y valor absoluto, traslaciones, expansión y contracción, optimización.}
\begin{multicols}{2}

\seccion{1. Esquema general}
\begin{enumerate}
  \item \textbf{Dominio:} dónde existe la fórmula (denominadores \(\ne0\), radicandos de índice par \(\ge0\), argumentos del logaritmo \(>0\)).
  \item \textbf{Cortes, simetría y signo:} con el eje \(Y\), \(x=0\); con el eje \(X\), \(f(x)=0\). Par si \(f(-x)=f(x)\) (simétrica respecto al eje \(Y\)); impar si \(f(-x)=-f(x)\) (respecto al origen).
  \item \textbf{Continuidad y límites} en los bordes del dominio.
  \item \textbf{Asíntotas:} vertical, horizontal u oblicua.
  \item \textbf{Derivada y monotonía:} signo de \(f'\).
  \item \textbf{Extremos relativos.}
  \item \textbf{Tabla y boceto} con toda la información.
\end{enumerate}

\seccion{2. Monotonía y extremos}
\begin{formula}
\(f'(x)>0\) en un intervalo \(\Rightarrow f\) crece; \quad \(f'(x)<0\Rightarrow f\) decrece.
\end{formula}
\begin{itemize}
  \item \textbf{Puntos críticos:} \(f'(a)=0\) o \(f'\) no existe.
  \item \textbf{Máximo relativo:} \(f'\) pasa de \(+\) a \(-\). \textbf{Mínimo:} de \(-\) a \(+\). Si no cambia de signo, no hay extremo.
  \item Con la segunda derivada: \(f'(a)=0\) y \(f''(a)<0\Rightarrow\) máximo; \(f''(a)>0\Rightarrow\) mínimo.
  \item \textbf{Curvatura} (ampliación): \(f''>0\), forma de cuenco; \(f''<0\), forma de arco. \textbf{Inflexión}: cambia la curvatura.
\end{itemize}
\begin{ejemplo}
\(f(x)=x^3-3x\): dominio \(\mathbb R\), impar, cortes en \(x=0,\pm\sqrt3\).
\(f'(x)=3x^2-3=0\Rightarrow x=\pm1\).
\[\begin{array}{c|ccccc} x & (-\infty,-1) & -1 & (-1,1) & 1 & (1,+\infty)\\\hline f' & + & 0 & - & 0 & +\\ f & \nearrow & 2 & \searrow & -2 & \nearrow\end{array}\]
Máximo \((-1,2)\), mínimo \((1,-2)\). \(f''(x)=6x\): inflexión en \((0,0)\).
\end{ejemplo}

\seccion{3. Asíntotas}
\begin{formula}
\textbf{Vertical} \(x=a\): \(\lim_{x\to a^\pm}f(x)=\pm\infty\).\\
\textbf{Horizontal} \(y=L\): \(\lim_{x\to\pm\infty}f(x)=L\).\\
\textbf{Oblicua} \(y=mx+n\):
\[m=\lim_{x\to\pm\infty}\frac{f(x)}{x}\ (\ne0),\qquad n=\lim_{x\to\pm\infty}\bigl(f(x)-mx\bigr)\]
\end{formula}
Se estudian por separado \(+\infty\) y \(-\infty\). La gráfica puede cortar una asíntota horizontal u oblicua; nunca una vertical.

\seccion{4. Polinómicas}
Dominio \(\mathbb R\), continuas y sin asíntotas. Un polinomio de grado \(n\) tiene como mucho \(n\) raíces y \(n-1\) extremos. Las ramas en \(\pm\infty\) las decide el término de mayor grado: \(a_nx^n\).
\sub{Raíces y signo.} Una raíz simple cruza el eje; una doble lo toca sin cruzarlo (cambia o no cambia el signo).

\seccion{5. Racionales \(f=\frac PQ\)}
\begin{formula}
\[D=\{x\in\mathbb R:\ Q(x)\ne0\}\]
\end{formula}
\begin{itemize}
  \item Si \(Q(a)=0\) y \(P(a)\ne0\): asíntota vertical \(x=a\).
  \item Si \(P(a)=Q(a)=0\): se simplifica; puede quedar un \textbf{hueco} (discontinuidad evitable) en lugar de una asíntota.
  \item Grado \(P<\) grado \(Q\): \(y=0\). Grados iguales: \(y=\frac{a_n}{b_n}\). Grado \(P=\) grado \(Q+1\): asíntota oblicua.
\end{itemize}
\begin{ejemplo}
\(f(x)=\dfrac{x^2+1}{x}=x+\dfrac1x\): AV \(x=0\), AO \(y=x\). \(f'(x)=\dfrac{x^2-1}{x^2}=0\Rightarrow x=\pm1\): máximo \((-1,-2)\) y mínimo \((1,2)\).
\end{ejemplo}

\seccion{6. Exponenciales \(y=a^x\) \((a>0,\ a\ne1)\)}
Dominio \(\mathbb R\), recorrido \((0,+\infty)\), pasa por \((0,1)\), asíntota horizontal \(y=0\). Si \(a>1\) crece; si \(0<a<1\) decrece.
\begin{formula}
\[V(t)=V_0(1+r)^t,\qquad r>-1\]
\(r>0\): crecimiento (interés compuesto). \(r<0\): depreciación.
\end{formula}
\begin{ejemplo}
Una máquina de 20\,000\,euros pierde el 15\,\% de su valor cada año: \(V(t)=20\,000\cdot0{,}85^{\,t}\). A los 3 años: \(V(3)\approx12\,282{,}50\)\,euros.
\end{ejemplo}

\seccion{7. Logarítmicas \(y=\log_ax\)}
\begin{formula}
\[\log_a x=y\iff a^y=x\]
\[\log_a(uv)=\log_a u+\log_a v,\qquad \log_a\frac uv=\log_a u-\log_a v\]
\[\log_a(u^r)=r\log_a u,\qquad \log_a x=\frac{\ln x}{\ln a}\]
\end{formula}
Dominio \((0,+\infty)\), recorrido \(\mathbb R\), pasa por \((1,0)\), asíntota vertical \(x=0\). Es la inversa de \(a^x\): gráficas simétricas respecto a \(y=x\).
\begin{ejemplo}
\(\log_2(x-1)=3\Rightarrow x-1=2^3\Rightarrow x=9\). Comprobación: \(x-1=8>0\) y \(\log_28=3\). En las ecuaciones con logaritmos hay que comprobar que cada candidato está en el dominio.
\end{ejemplo}
\begin{ejemplo}
\textquestiondown{}Cuándo se duplica un capital al 5\,\% anual? \(2=1{,}05^{\,t}\Rightarrow t=\dfrac{\ln2}{\ln1{,}05}\approx14{,}2\) años.
\end{ejemplo}

\seccion{8. A trozos y valor absoluto}
\begin{formula}
\[|x|=\begin{cases}-x,&x<0\\ x,&x\ge0\end{cases}\]
\end{formula}
Para \(|g(x)|\): se estudia el signo de \(g\) y se escribe por trozos. En cada punto de unión se comprueban los laterales: si coinciden con el valor, no hay salto; si además coinciden las derivadas laterales, no hay pico.
\begin{ejemplo}
\(g(x)=|x-1|-2\) vale \(-x-1\) si \(x<1\) y \(x-3\) si \(x\ge1\): una \guillemotleft{}V\guillemotright{} con vértice en \((1,-2)\) que corta el eje \(X\) en \(x=-1\) y \(x=3\).
\end{ejemplo}

\seccion{9. Transformaciones de la gráfica de \(f\)}
\begin{formula}
\textbf{Traslación:}
\[g(x)=f(x-a)+k:\qquad (u,\,f(u))\mapsto(u+a,\ f(u)+k)\]
\textbf{Escala:}
\[g(x)=a\,f(bx):\qquad (u,\,v)\mapsto(u/b,\ av)\]
\end{formula}
\begin{itemize}
  \item \(a>0\) en \(f(x-a)\): a la \textbf{derecha}; \(k>0\): hacia \textbf{arriba}.
  \item \(|a|>1\) estira en vertical; \(|b|>1\) comprime en horizontal.
  \item \(-f(x)\): simetría respecto al eje \(X\). \(f(-x)\): respecto al eje \(Y\).
  \item Las asíntotas y los extremos se transforman con la misma regla.
\end{itemize}
\begin{ejemplo}
\(g(x)=(x-2)^2-3\) es \(y=x^2\) trasladada 2 a la derecha y 3 abajo: vértice \((2,-3)\).
\(r(x)=\frac1{x+2}+1\): asíntotas \(x=-2\) e \(y=1\).
\end{ejemplo}

\seccion{10. Optimización}
\begin{enumerate}
  \item Elige la variable y escribe la función que se optimiza.
  \item Fija el \textbf{dominio} con las restricciones del enunciado.
  \item Deriva, iguala a 0 y obtén los candidatos interiores.
  \item Compara con los \textbf{extremos del intervalo}.
  \item Responde con unidades.
\end{enumerate}
\begin{ejemplo}
\(B(q)=-q^2+40q-300\) miles de euros, \(q\in[0,30]\) toneladas. \(B'(q)=-2q+40=0\Rightarrow q=20\), \(B(20)=100\). Si la capacidad fuera \(q\le15\), \(B\) crece en \([0,15]\) y el máximo es \(B(15)=75\).
\end{ejemplo}
\begin{ejemplo}
\textbf{Coste medio.} \(C(q)=q^2+100\) euros por \(q\) unidades. \(\overline C(q)=\dfrac{C(q)}q=q+\dfrac{100}q\), \(q>0\).
\(\overline C'(q)=1-\dfrac{100}{q^2}=0\Rightarrow q=10\); \(\overline C'\) pasa de \(-\) a \(+\): mínimo, \(\overline C(10)=20\)\,euros/unidad.
\end{ejemplo}
\begin{ejemplo}
\textbf{Geometría.} Rectángulo de perímetro 40 m y área máxima: lados \(x\) y \(20-x\), \(A(x)=x(20-x)\), \(x\in(0,20)\). \(A'(x)=20-2x=0\Rightarrow x=10\): un cuadrado de 100 m\(^2\).
\end{ejemplo}

\begin{ojo}
\begin{itemize}
  \item Concluir que hay asíntota vertical sin comprobar que el numerador no se anula.
  \item Olvidar el dominio del logaritmo o del problema.
  \item Confundir el sentido: \(f(x-2)\) se mueve a la derecha.
  \item Dar un extremo sin comprobar el cambio de signo de \(f'\).
  \item En optimización, no comparar con los extremos del intervalo.
\end{itemize}
\end{ojo}
