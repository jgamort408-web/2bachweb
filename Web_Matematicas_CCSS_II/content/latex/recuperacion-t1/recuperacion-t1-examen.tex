\section*{Ejercicio 1 \hfill 4 puntos}
\textit{Tiempo orientativo: 15 minutos.}

Se considera la función
\[
f(x)=
\begin{cases}
ax-3x^2,& x\leq 1,\\
2x^2+b,& x>1.
\end{cases}
\]
\begin{enumerate}[label=\textbf{(\alph*)},leftmargin=*,itemsep=1em]
  \item \textbf{(2 puntos)} Calcula los valores de \(a\) y \(b\) para que
  \(f\) sea derivable en \(x=1\).
  \item \textbf{(2 puntos)} Para \(a=3\) y \(b=-2\), halla la ecuación de la
  \textbf{recta tangente} a la gráfica de \(f\) en el punto de abscisa \(x=2\).
\end{enumerate}

\section*{Ejercicio 2 \hfill 6 puntos}
\textit{Tiempo orientativo: 40 minutos.}

Dada la función
\[
g(x)=\frac{1}{x+3},
\]
\begin{enumerate}[label=\textbf{(\alph*)},leftmargin=*,itemsep=1em]
  \item \textbf{(4 puntos)} Estudia la continuidad, las asíntotas,
  la monotonía y la curvatura de \(g\), y represéntala gráficamente.
  \item \textbf{(2 puntos)} Calcula el área comprendida entre la gráfica
  de \(g\), el eje de abscisas y las rectas \(x=2\) y \(x=4\).
\end{enumerate}
